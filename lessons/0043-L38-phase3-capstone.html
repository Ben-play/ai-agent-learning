Phase 3 综合实战：可验收的 Agent 客户端

先证明工具、失败分支与成本真的接起来，再接原生模型接口

💡 为什么这课重要？

L25–L37 给出了接口、流式、工具、重试、上下文和成本的零件。本课不再凭一段正确答案宣告“生产级”，而是构建一个完全离线、可复测的执行骨架：两次模型往返真的产生一个工具结果，失败不能伪装成成功，账本也不能一直为零。

两种 fixture adapter 都是虚构的本地响应器，不联网、不读取密钥、不代表两家真实厂商。它们只验证应用控制逻辑。真实模型能力、原生缓存命中、SSE网络流、SDK兼容与线上延迟，需要独立的、有预算的集成测试。

学习目标与分段安排

  1. 第一段：定义请求/响应边界、终态和成本账本，理解两次往返为什么不能少记一次。
  2. 第二段：接入有界重试、Fallback、执行权限和工具结果；拒答/截断不走通用重试。
  3. 第三段：分别验收 text/json/本地流式消费、异常和预算，并设计真实模型评估。

每段都可以暂停。完成自检后，隔一天不看答案重建“拒答”“工具失败”和“未知用量”三个分支，避免只记得复制过程。

2. 完整离线骨架：一份代码，一种消息表示

下面整块保存为 agent_lab.py，再接上下一节自检；仅用 Python 标准库。先看三条不变量：每次底层尝试入账、工具先授权再执行、每个请求 ID 都有结果。价格是虚构计价单位，不是美元报价。

from copy import deepcopy
from dataclasses import dataclass
from decimal import Decimal
import json
import random
import time


class TransientError(RuntimeError):
    """仅用于本实验的临时传输故障；不表示所有异常都可重试。"""


class BudgetExceeded(RuntimeError):
    pass


@dataclass(frozen=True)
class Usage:
    # 四项互不重叠；真实 SDK 字段若有包含关系，adapter 必须先转换。
    fresh: int = 0
    write: int = 0
    read: int = 0
    output: int = 0

    def __post_init__(self):
        values = (self.fresh, self.write, self.read, self.output)
        if any(type(n) is not int or n < 0 for n in values):
            raise ValueError("usage must contain non-negative integers")


@dataclass(frozen=True)
class ToolCall:
    id: str
    name: str
    arguments: str


@dataclass(frozen=True)
class Reply:
    content: str = ""
    calls: tuple = ()
    status: str = "completed"  # 本地契约，不是所有厂商的字段枚举
    usage: Usage | None = None


class Ledger:
    def __init__(self):
        self.rows = []

    def record(self, provider, usage, status):
        # 示例费率：每百万单位输入 fresh/write/read/output 分别为 2/3/0.2/8。
        # 不按‘缓存读是固定几折’猜真实厂商价格。
        cost = None
        if usage is not None:
            cost = (Decimal(usage.fresh) * 2 + Decimal(usage.write) * 3
                    + Decimal(usage.read) * Decimal("0.2")
                    + Decimal(usage.output) * 8) / 1_000_000
        self.rows.append({"provider": provider, "status": status,
                          "usage": usage, "cost": cost})

    def report(self):
        return {
            "attempts": len(self.rows),
            "known_cost": sum((r["cost"] for r in self.rows
                               if r["cost"] is not None), Decimal(0)),
            "unknown_usage": sum(r["usage"] is None for r in self.rows),
        }


class Tools:
    def __init__(self, fail=False):
        self.executed = []  # 可观察的环境结果，不只看最终回答
        self.fail = fail

    def dispatch(self, call, allowed):
        if call.name not in allowed:
            return {"ok": False, "error": "not_allowed"}
        if call.name != "get_weather":
            return {"ok": False, "error": "unknown_tool"}
        try:
            args = json.loads(call.arguments)
        except (json.JSONDecodeError, TypeError):
            return {"ok": False, "error": "invalid_json"}
        if (type(args) is not dict or set(args) != {"city"}
                or type(args["city"]) is not str or not args["city"].strip()):
            return {"ok": False, "error": "invalid_arguments"}
        # 只有通过授权与校验后才进入真实执行接缝。
        self.executed.append((call.name, args["city"]))
        if self.fail:
            return {"ok": False, "error": "tool_unavailable"}
        return {"ok": True, "data": {"city": args["city"], "temperature": 22}}


class FixtureAdapter:
    """本地假模型；不伪造SDK、联网或真实缓存。"""
    def __init__(self, mode="normal", failures=0, cached=False):
        self.mode = mode
        self.failures = failures
        self.cached = cached

    def __call__(self, history, output):
        if self.failures:
            self.failures -= 1
            raise TransientError("fixture transport unavailable")
        usage = Usage(fresh=20, read=100, output=30) if self.cached else Usage(fresh=120, output=30)
        if self.mode == "refused":
            return Reply(status="refused", usage=usage)
        if self.mode == "incomplete":
            return Reply(content="half", status="incomplete", usage=usage)
        results = [m for m in history if m["role"] == "tool"]
        if not results:
            name = "other" if self.mode == "unknown_tool" else "get_weather"
            args = '{"city": 7}' if self.mode == "bad_args" else '{"city": "北京"}'
            return Reply(calls=(ToolCall("call-1", name, args),), usage=usage)
        result = json.loads(results[-1]["content"])
        text = "北京现在22°C。" if result["ok"] else "工具未成功，不能提供天气。"
        if self.mode == "bad_json":
            text = "not-json"
        elif output == "json":
            text = json.dumps({"summary": text}, ensure_ascii=False)
        return Reply(content=text, usage=usage)


class AgentClient:
    def __init__(self, adapters, tools, *, max_calls=8, max_turns=4, wait=time.sleep):
        if not adapters or any(type(n) is not int or n <= 0 for n in (max_calls, max_turns)):
            raise ValueError("adapters and positive budgets required")
        self.adapters = tuple(adapters)
        self.tools = tools
        self.max_calls = max_calls
        self.max_turns = max_turns
        self.wait = wait
        self.ledger = Ledger()

    def _call(self, history, output):
        last = None
        for name, adapter in self.adapters:
            for attempt in range(2):  # 每个provider最多2次尝试，只有此处拥有重试
                if len(self.ledger.rows) >= self.max_calls:
                    raise BudgetExceeded("request attempt budget exhausted")
                try:
                    reply = adapter(deepcopy(history), output)
                except TransientError as exc:
                    self.ledger.record(name, None, "transient_error")
                    last = exc
                    if attempt == 0:
                        self.wait(random.uniform(0, 0.1))  # full jitter；自检注入不等待
                    continue
                except Exception:
                    # 只记未知用量并原样抛出；不是可重试异常，不切厂商。
                    self.ledger.record(name, None, "non_retryable_error")
                    raise
                self.ledger.record(name, reply.usage, reply.status)
                return reply
        raise TransientError("all fixture adapters unavailable") from last

    def generate(self, prompt, *, output="text", allowed=frozenset({"get_weather"})):
        if output not in {"text", "json", "stream"}:
            raise ValueError("unknown output mode")
        self.ledger = Ledger()  # 每个任务一份账本；重用实例不混账
        self.history = [{"role": "user", "content": prompt}]
        for _ in range(self.max_turns):
            reply = self._call(self.history, output)
            if reply.status != "completed":
                # 拒答或截断不靠切厂商绕过去，也不执行半截工具参数。
                raise ValueError(f"not completed: {reply.status}")
            if reply.calls:
                ids = [call.id for call in reply.calls]
                if any(not i for i in ids) or len(ids) != len(set(ids)):
                    raise ValueError("invalid tool call ids")
                self.history.append({"role": "assistant", "calls": reply.calls})
                for call in reply.calls:
                    result = self.tools.dispatch(call, allowed)
                    self.history.append({"role": "tool", "call_id": call.id,
                                         "content": json.dumps(result, ensure_ascii=False)})
                continue
            self.history.append({"role": "assistant", "content": reply.content})
            if output == "json":
                data = json.loads(reply.content)  # 本地验证，不冒充约束解码
                if type(data) is not dict or set(data) != {"summary"} or type(data["summary"]) is not str:
                    raise ValueError("invalid final schema")
                return data
            if output == "stream":
                # 已取得完整fixture响应再切片：只验证消费者，不声称改善真实TTFT。
                return [("delta", reply.content[:4]), ("delta", reply.content[4:]),
                        ("done", None)]
            return reply.content
        raise BudgetExceeded("tool round budget exhausted")


def consume(events):
    parts = []
    for kind, value in events:
        if kind == "delta":
            parts.append(value)
        elif kind == "done":
            return "".join(parts)
        elif kind == "error":
            raise ValueError("stream failed")
    raise ValueError("missing terminal event")

读代码的顺序：从 generate 找调用点 → 看 _call 的尝试预算与入账 → 看 dispatch 的零执行失败分支 → 再看 fixture 如何决定下一轮。None 用量表示未知，不是零费用；报告只有已知成本的小计。

def make(mode="normal", **kwargs):
    tools = Tools()
    return AgentClient([("fixture-a", FixtureAdapter(mode))], tools,
                       wait=lambda _: None, **kwargs)


def must_fail(fn, error):
    try:
        fn()
    except error:
        return
    raise AssertionError(f"expected {error.__name__}")


client = make()
assert client.generate("查天气") == "北京现在22°C。"
assert client.tools.executed == [("get_weather", "北京")]
assert [m["role"] for m in client.history] == ["user", "assistant", "tool", "assistant"]
assert client.history[1]["calls"][0].id == client.history[2]["call_id"]
assert client.ledger.report() == {
    "attempts": 2, "known_cost": Decimal("0.00096"), "unknown_usage": 0
}
assert make().generate("查天气", output="json") == {"summary": "北京现在22°C。"}
assert consume(make().generate("查天气", output="stream")) == "北京现在22°C。"
must_fail(lambda: consume([("delta", "半句")]), ValueError)
must_fail(lambda: consume([("error", None)]), ValueError)

for mode in ("refused", "incomplete"):
    client = make(mode)
    must_fail(lambda: client.generate("查天气"), ValueError)
    assert client.ledger.report()["attempts"] == 1
    assert client.tools.executed == []

for mode in ("bad_args", "unknown_tool"):
    client = make(mode)
    assert "未成功" in client.generate("查天气")
    assert client.tools.executed == []
client = make()
assert "未成功" in client.generate("查天气", allowed=frozenset())
assert client.tools.executed == []
client = AgentClient([("fixture-a", FixtureAdapter())], Tools(fail=True), wait=lambda _: None)
assert "未成功" in client.generate("查天气")
assert len(client.tools.executed) == 1
must_fail(lambda: make("bad_json").generate("查天气", output="json"), ValueError)

client = AgentClient([("fixture-a", FixtureAdapter(failures=2)),
                      ("fixture-b", FixtureAdapter())], Tools(), wait=lambda _: None)
assert client.generate("查天气") == "北京现在22°C。"
assert client.ledger.report()["attempts"] == 4
assert client.ledger.report()["unknown_usage"] == 2  # 不能把失败请求当免费
assert [r["provider"] for r in client.ledger.rows] == ["fixture-a", "fixture-a", "fixture-b", "fixture-a"]

def invalid_adapter(history, output):
    raise ValueError("fixture non-retryable configuration error")

client = AgentClient([("invalid", invalid_adapter), ("backup", FixtureAdapter())],
                     Tools(), wait=lambda _: None)
must_fail(lambda: client.generate("查天气"), ValueError)
assert client.ledger.report() == {
    "attempts": 1, "known_cost": Decimal(0), "unknown_usage": 1
}
assert client.tools.executed == []  # 不重试、不切backup、不产生工具动作

client = make(max_calls=1)
must_fail(lambda: client.generate("查天气"), BudgetExceeded)
assert client.ledger.report()["attempts"] == 1
must_fail(lambda: make(max_turns=1).generate("查天气"), BudgetExceeded)

# 计费字段的算法对照，不证明真实缓存生效。
ledger = Ledger()
ledger.record("fixture", Usage(fresh=20, write=100, output=30), "completed")
ledger.record("fixture", Usage(fresh=20, read=100, output=30), "completed")
assert ledger.report()["known_cost"] == Decimal("0.00088")

特别观察 Fallback：fixture-a 前两次临时失败后，fixture-b 完成当前调用；下一轮重新从首选开始。因此总共四次底层尝试，不能把一次业务任务、一次工具执行和一次模型尝试混成同一个计数。

怎么看失败：断言告诉你哪条不变量被破坏，不是让你修改预期去迎合错误代码。工具异常只反馈受控错误码，不输出完整输入、凭证或堆栈。

4. 从确定性自检走向 Agent 评估

Task 是有输入与成功标准的任务；trial 是一次尝试；trace 是执行记录；outcome 是环境最终状态。回答“执行成功”不能代替检查环境结果。上述 executed 和配对记录是确定性检查；真实任务不应要求所有正确解法都走固定调用顺序。

  1. 建立正常、无权限、缺参数、工具失败等小任务集，明确环境结果与成功门槛。
  2. 接真实模型后，对同一任务重复试验，分别报告成功率、样本数、成本和延迟；fixture 不能证明模型稳定。
  3. 优先使用可判定的状态检查；需要 Judge 时先校准人工一致性，并隔离开发集与测试集。
  4. 将每个真实故障变成回归用例；规则可以增删，是否保留由回归结果判断。

依据：Demystifying Evals（推荐精读）。真实调用需要先确认预算；本页没有自动联网开关。

Q 工具返回失败时，什么才是正确的验收态度？
执行意图、调用尝试和环境结果是三件事。受控失败是正确的控制行为，但不代表业务任务完成。

5. 连接真实接口前的检查清单

  • 协议：Responses 的 function_call_output/call_id、Messages 的 tool_result/tool_use_id、Chat 的 tool/tool_call_id 分别实现，不跨协议照抄。
  • 原生状态：Adapter 内保留完整 Items/内容块；不能只存本课的文本投影或修改不透明推理内容。
  • 输出：结构化约束针对受支持接口中正常完成的响应；拒答、截断和业务错误单独处理。本地JSON验证不是约束解码。结构约束也不要求额外再生成一遍最终答案。
  • 重试：本课只重试 TransientError；真实接入映射具体异常，设单一重试 owner 与墙钟预算，不能同时叠 SDK 和外层无限重试。
  • 流式：真实流要处理工具参数增量、终态、取消和未知用量；本课本地切片不模拟真实TTFT。
  • 成本：拆清输入包含关系、缓存写/读、输出与工具费用；没有usage时保留未知，不能宣称账单为零或降本百分比。
  • 上线：还需 L30 上下文守卫、持久执行、租户权限、外部动作幂等、脱敏观测和部署回归；不因本地通过就宣称生产可用。
Q 传输失败且拿不到 usage 时，应怎样记账？
已知成本只是小计；未知不等于免费。生产需结合provider可用的请求标识和账单数据对账。
